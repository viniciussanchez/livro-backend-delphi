rem ---- 17.3 Um servidor Linux de laboratório
docker run --rm -v "%cd%\event-api\Linux64\Release:/opt/eventos/event-api:ro" ubuntu:22.04 ldd /opt/eventos/event-api/EventApi
docker run --rm --network cap17-linux_default -e DB_HOST=postgres -v "%cd%\event-api\Linux64\Release:/opt/eventos/event-api:ro" ubuntu:22.04 /opt/eventos/event-api/EventApi
docker build -t eventos-servidor-linux -f linux\Dockerfile.servidor linux
docker run -d --init --name eventos-linux --network cap17-linux_default -p 9000-9003:9000-9003 -e DB_HOST=postgres -v "%cd%\event-api\Linux64\Release:/opt/eventos/event-api:ro" -v "%cd%\customer-api\Linux64\Release:/opt/eventos/customer-api:ro" -v "%cd%\ticket-api\Linux64\Release:/opt/eventos/ticket-api:ro" -v "%cd%\api-gateway\Linux64\Release:/opt/eventos/api-gateway:ro" -v "%cd%\linux:/opt/eventos/scripts:ro" eventos-servidor-linux sleep infinity
docker exec eventos-linux sh /opt/eventos/scripts/start-all.sh
