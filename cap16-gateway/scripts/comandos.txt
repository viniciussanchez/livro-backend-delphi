rem login: copie o accessToken da resposta (sem as aspas) para a variavel abaixo
curl -X POST http://localhost:9000/auth/token -H "Content-Type: application/json" -d "{\"email\":\"admin@eventos.local\",\"password\":\"admin12345\"}"
set ADMIN=cole-aqui-o-token
curl -s -o nul -w "venue %{http_code} " -X POST http://localhost:9000/venues -H "Authorization: Bearer %ADMIN%" -H "Content-Type: application/json" -d "{\"name\":\"Arena do Gateway\",\"address\":\"Rua A, 1\",\"city\":\"BH\",\"state\":\"MG\",\"capacity\":500}"
curl -s -o nul -w "evento %{http_code} " -X POST http://localhost:9000/events -H "Authorization: Bearer %ADMIN%" -H "Content-Type: application/json" -d "{\"name\":\"Show do Gateway\",\"categoryId\":1,\"venueId\":1,\"startAt\":\"2099-03-10T20:00:00\"}"
curl -s -o nul -w "lote %{http_code} " -X POST http://localhost:9000/events/1/batches -H "Authorization: Bearer %ADMIN%" -H "Content-Type: application/json" -d "{\"name\":\"Pista\",\"price\":120,\"quantity\":50,\"salesStart\":\"2020-01-01T00:00:00\",\"salesEnd\":\"2099-03-01T00:00:00\"}"
curl -s -o nul -w "publica %{http_code} " -X POST http://localhost:9000/events/1/publish -H "Authorization: Bearer %ADMIN%"
curl -s -o nul -w "cliente %{http_code}" -X POST http://localhost:9000/customers -H "Content-Type: application/json" -d "{\"cpf\":\"529.982.247-25\",\"name\":\"Ana Souza\",\"email\":\"ana@exemplo.com\",\"password\":\"senha-da-ana\"}"
